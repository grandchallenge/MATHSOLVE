        pairs = (A[i][j], A[i][k], A[i][l], A[j][k], A[j][l], A[k][l])
        if all(x == 1 for x in pairs):
            red_num += mass
        elif all(x == 0 for x in pairs):
            blue_num += mass
    denom = sum(w) ** 4
    density = Fraction(red_num + blue_num, denom)
    return {
        "red_num": red_num,
        "blue_num": blue_num,
        "denom": denom,
        "density": density,
        "reference_beaten": int(density < B_STAR),
        "density_ppt": (PPT_SCALE * density.numerator + density.denominator - 1) // density.denominator,
    }


def partition_collision_counter(weights: list[int], red_rows: list[str], normalize_gcd: bool = False):
    w, rows = validate_template(weights, red_rows, normalize_gcd=normalize_gcd)
    m = len(w)
    breakdown = {}
    totals = {}
    for color_name, color_char in (("red", "1"), ("blue", "0")):
        L = [1 if rows[i][i] == color_char else 0 for i in range(m)]
        E = [[1 if (i != j and rows[i][j] == color_char) else 0 for j in range(m)] for i in range(m)]
        c_4 = sum((w[a] ** 4) * L[a] for a in range(m))
        c_31 = c_22 = c_211 = c_1111 = 0
        for a in range(m):
            for b in range(a + 1, m):
                if not E[a][b]:
                    continue
                c_31 += 4 * ((w[a] ** 3) * w[b] * L[a] + w[a] * (w[b] ** 3) * L[b])
                if L[a] and L[b]:
                    c_22 += 6 * (w[a] ** 2) * (w[b] ** 2)
                for d in range(b + 1, m):
                    if not (E[a][d] and E[b][d]):
                        continue
                    c_211 += 12 * w[a] * w[b] * w[d] * (w[a] * L[a] + w[b] * L[b] + w[d] * L[d])
                    for e in range(d + 1, m):
                        if E[a][e] and E[b][e] and E[d][e]:
                            c_1111 += 24 * w[a] * w[b] * w[d] * w[e]
        by_type = {(4,): c_4, (3, 1): c_31, (2, 2): c_22, (2, 1, 1): c_211, (1, 1, 1, 1): c_1111}
        breakdown[color_name] = by_type
        totals[color_name] = sum(by_type.values())
    denom = sum(w) ** 4
    return {
        "red_num": totals["red"],
        "blue_num": totals["blue"],
        "denom": denom,
        "density": Fraction(totals["red"] + totals["blue"], denom),
        "breakdown": breakdown,
    }


def delta_diag_contribution(weights: list[int], red_rows: list[str], i: int, color_char: str) -> int:
    w, rows = validate_template(weights, red_rows)
    m = len(w)
    L = [1 if rows[u][u] == color_char else 0 for u in range(m)]
    nbrs = [j for j in range(m) if j != i and rows[i][j] == color_char]
    w1 = sum(w[j] for j in nbrs)
    w2_loop = sum((w[j] ** 2) * L[j] for j in nbrs)
    w_edge = sum(
        w[nbrs[a]] * w[nbrs[b]]
        for a in range(len(nbrs))
        for b in range(a + 1, len(nbrs))
        if rows[nbrs[a]][nbrs[b]] == color_char
    )
    wi = w[i]
    return (wi**4) + 4 * (wi**3) * w1 + 6 * (wi**2) * w2_loop + 12 * (wi**2) * w_edge


def delta_edge_contribution(weights: list[int], red_rows: list[str], i: int, j: int, color_char: str) -> int:
    w, rows = validate_template(weights, red_rows)
    m = len(w)
    L = [1 if rows[u][u] == color_char else 0 for u in range(m)]
    common = [k for k in range(m) if k != i and k != j and rows[i][k] == color_char and rows[j][k] == color_char]
    wi, wj, li, lj = w[i], w[j], L[i], L[j]
    t_31 = 4 * wi * wj * ((wi**2) * li + (wj**2) * lj)
    t_22 = 6 * (wi**2) * (wj**2) * li * lj
    t_211 = 12 * wi * wj * sum(w[k] * (wi * li + wj * lj + w[k] * L[k]) for k in common)
    t_1111 = sum(
        24 * wi * wj * w[common[a]] * w[common[b]]
        for a in range(len(common))
        for b in range(a + 1, len(common))
        if rows[common[a]][common[b]] == color_char
    )
    return t_31 + t_22 + t_211 + t_1111
